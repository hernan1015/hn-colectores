# HN AGRO · firmador de enlaces R2 en la nube. Ver firmador-r2/LEEME.md.
# PARA INSTALARLO: copiar este archivo a .github/workflows/firmador-r2.yml (el token con el que se armó la rama no
# tiene permiso «workflow», por eso viene como .txt).
# SIN HORARIO A PROPÓSITO: lo dispara Supabase una vez por día (pg_cron + pg_net, como el radar) o una persona desde
# Actions → firmador-r2 → Run workflow. No se agrega schedule acá.
name: firmador-r2
on:
  workflow_dispatch:
    inputs:
      dias_rayos: { description: 'días de horas de rayos-hora a firmar (3 = 24 h de la ficha + 48 h de margen; 1..7)', required: false, default: '3' }
  repository_dispatch:
    types: [firmar-r2-enlaces]
permissions:
  contents: read
concurrency:
  group: firmador-r2
  cancel-in-progress: false
jobs:
  firmar:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - run: npm ci --prefix firmador-r2 --omit=dev --no-audit --no-fund
      - name: Firmar y escribir r2_enlaces
        env:
          R2_ACCOUNT_ID: ${{ secrets.R2_ACCOUNT_ID }}
          R2_ACCESS_KEY_ID: ${{ secrets.R2_ACCESS_KEY_ID }}
          R2_SECRET_ACCESS_KEY: ${{ secrets.R2_SECRET_ACCESS_KEY }}
          R2_BUCKET: ${{ secrets.R2_BUCKET }}
          SUPABASE_FIRMADOR_URL: ${{ secrets.SUPABASE_FIRMADOR_URL }}
          SUPABASE_FIRMADOR_CA: ${{ secrets.SUPABASE_FIRMADOR_CA }}   # OBLIGATORIO: sin la CA no se conecta (TLS verificado)
          DIAS_RAYOS: ${{ inputs.dias_rayos || github.event.client_payload.dias_rayos || '3' }}
        run: node firmador-r2/firmar-r2.mjs
