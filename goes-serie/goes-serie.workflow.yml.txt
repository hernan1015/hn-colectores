# HN AGRO · serie de imágenes GOES-19 FDCF en R2 (goes-serie/, 30 días rotando). Ver goes-serie/LEEME.md.
# PARA INSTALARLO: copiar este archivo a .github/workflows/goes-serie.yml (el token con el que se armó la rama no
# tiene permiso «workflow», por eso viene como .txt). Tiene que estar en main para workflow_dispatch.
# SIN HORARIO a propósito: lo activa local. Primero una corrida a mano (Actions → goes-serie → Run workflow).
# El disparo cada 10 min va desde Supabase (pg_cron + pg_net → workflow_dispatch, como el radar: ver LEEME).
# Si se prefiere el cron de Actions, sería:  schedule: - cron: '*/10 * * * *'  (GitHub lo atrasa y a veces lo saltea;
# con HORAS=3 cada corrida se pone al día con lo que haya quedado atrás).
name: goes-serie
on:
  workflow_dispatch:
    inputs:
      horas: { description: 'ventana hacia atrás, en horas (3; de 1 a 24)', required: false, default: '3' }
  repository_dispatch:
    types: [goes-serie]
permissions:
  contents: read
concurrency:
  group: goes-serie            # la misma que goes-evento: nunca corren a la vez (el índice se lee y se reescribe)
  cancel-in-progress: false
jobs:
  serie:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - name: Serie GOES
        env:
          R2_ACCOUNT_ID: ${{ secrets.R2_ACCOUNT_ID }}
          R2_ACCESS_KEY_ID: ${{ secrets.R2_ACCESS_KEY_ID }}
          R2_SECRET_ACCESS_KEY: ${{ secrets.R2_SECRET_ACCESS_KEY }}
          R2_BUCKET: ${{ secrets.R2_BUCKET }}
          MODO: serie
          HORAS: ${{ inputs.horas || github.event.client_payload.horas || '3' }}
        run: node goes-serie/goes-serie.mjs
