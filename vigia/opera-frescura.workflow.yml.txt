# HN AGRO · VIGÍA DE FRESCURA DE LOS MOSAICOS OPERA (DISEÑO: APAGADO, no se prende sin el OK de local y el maker).
# Va en .github/workflows/opera-frescura.yml (copiarlo tal cual desde vigia/opera-frescura.workflow.yml.txt; el token de pc-nanci no tiene el permiso `workflow`, igual que con el vigía de SATCHI).
# Una vez por día lee el índice PÚBLICO de pasadas OPERA de SATCHI (satchi-publico?que=opera&zona=corrientes&pasadas=1,
# sin llaves) y manda UN mail si `hoy − hasta` > 3 días, si `fuente` no es `r2` o si responde `ok:false`.
# NO REPITE: la clave lleva `hasta` y se guarda en la caché de Actions sólo DESPUÉS de mandar el mail.
# El mail usa los MISMOS secretos que el vigía de SATCHI: SATCHI_SMTP_HOST/PORT/USER/PASS (FROM opcional) y SATCHI_VIGIA_DESTINO.
# Detalle: vigia/opera-frescura.mjs y hn-cortex docs/opera-mosaicos-2026-11-11/DIAGNOSTICO.md §7.
# El registro de Actions es PÚBLICO: el script imprime sólo el veredicto y la clave (datos públicos).
name: opera-frescura
on:
  # APAGADO: el horario va comentado. Para prenderlo, descomentar y sacar el `if: false` del job.
  # schedule:
  #   - cron: '17 12 * * *'    # una vez por día, 12:17 UTC (09:17 ART), lejos de la hora en punto
  workflow_dispatch:
permissions:
  contents: read
concurrency:
  group: opera-frescura
  cancel-in-progress: false
jobs:
  frescura:
    if: false                  # APAGADO también a mano, hasta el OK
    runs-on: ubuntu-latest
    timeout-minutes: 3
    steps:
      - uses: actions/checkout@v4
        with:
          sparse-checkout: vigia
      - name: Tests (sin red ni mail)
        run: node --test vigia/test/opera-frescura.test.mjs
      - name: Evaluar el índice público
        id: ev
        run: node vigia/opera-frescura.mjs evaluar
      - name: ¿Ya se avisó esta clave?
        if: steps.ev.outputs.avisar == 'si'
        id: ya
        uses: actions/cache/restore@v4
        with:
          path: .opera-frescura-aviso
          key: opera-frescura-${{ steps.ev.outputs.clave }}
          lookup-only: true
      - name: Mandar el mail
        if: steps.ev.outputs.avisar == 'si' && steps.ya.outputs.cache-hit != 'true'
        env:
          SATCHI_SMTP_HOST: ${{ secrets.SATCHI_SMTP_HOST }}
          SATCHI_SMTP_PORT: ${{ secrets.SATCHI_SMTP_PORT }}
          SATCHI_SMTP_USER: ${{ secrets.SATCHI_SMTP_USER }}
          SATCHI_SMTP_PASS: ${{ secrets.SATCHI_SMTP_PASS }}
          SATCHI_SMTP_FROM: ${{ secrets.SATCHI_SMTP_FROM }}
          SATCHI_VIGIA_DESTINO: ${{ secrets.SATCHI_VIGIA_DESTINO }}
          CLAVE: ${{ steps.ev.outputs.clave }}
        run: node vigia/opera-frescura.mjs mandar --clave "$CLAVE"
      - name: Recordar la clave (sólo si el mail salió)
        if: steps.ev.outputs.avisar == 'si' && steps.ya.outputs.cache-hit != 'true' && hashFiles('.opera-frescura-aviso/clave') != ''
        uses: actions/cache/save@v4
        with:
          path: .opera-frescura-aviso
          key: opera-frescura-${{ steps.ev.outputs.clave }}
